cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" KVR_RELEASE_VERSION LIMIT_COUNT 1)
project(KotorVR VERSION ${KVR_RELEASE_VERSION} LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "KOTOR requires an x86 build")
endif()
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_API_LAYERS OFF CACHE BOOL "" FORCE)
set(BUILD_CONFORMANCE_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_LOADER ON CACHE BOOL "" FORCE)
set(DYNAMIC_LOADER OFF CACHE BOOL "" FORCE)
set(INSTALL_TO_ARCHITECTURE_PREFIXES OFF CACHE BOOL "" FORCE)
add_subdirectory(deps/OpenXR-SDK EXCLUDE_FROM_ALL)
add_subdirectory(deps/minhook EXCLUDE_FROM_ALL)
set(KVR_OUTPUT_DIRECTORY "${PROJECT_SOURCE_DIR}/bin" CACHE PATH "Runtime output directory")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${KVR_OUTPUT_DIRECTORY}")
configure_file(src/version.h.in "${PROJECT_BINARY_DIR}/generated/kvr_version.h" @ONLY)
configure_file(src/version.rc.in "${PROJECT_BINARY_DIR}/generated/version.rc" @ONLY)
include_directories("${PROJECT_BINARY_DIR}/generated")
set(KVR_VERSION_RESOURCE "${PROJECT_BINARY_DIR}/generated/version.rc")
option(KVR_BUILD_GAME "Build the injected game module" ON)
if(KVR_BUILD_GAME)
  add_library(kotor_vr SHARED src/game_mod.cpp src/vr_backend.cpp src/hud_overlay.cpp src/vr_cursor.cpp ${KVR_VERSION_RESOURCE})
  target_compile_definitions(kotor_vr PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_OPENGL)
  target_include_directories(kotor_vr PRIVATE deps/OpenXR-SDK/include deps/minhook/include)
  target_link_libraries(kotor_vr PRIVATE openxr_loader minhook opengl32 glu32 user32 gdi32 bcrypt)
  target_compile_definitions(kotor_vr PRIVATE KVR_FILE_TYPE=2)
endif()
add_executable(kotor-vr-launcher src/launcher.cpp ${KVR_VERSION_RESOURCE})
target_compile_definitions(kotor-vr-launcher PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(kotor-vr-launcher PRIVATE bcrypt user32)
add_executable(kotor-vr-probe src/vr_probe.cpp src/vr_backend.cpp ${KVR_VERSION_RESOURCE})
target_include_directories(kotor-vr-probe PRIVATE deps/OpenXR-SDK/include)
target_compile_definitions(kotor-vr-probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(kotor-vr-probe PRIVATE openxr_loader opengl32 user32 gdi32)
add_executable(kotor-vr-math-test tests/vr_math_test.cpp ${KVR_VERSION_RESOURCE})
target_include_directories(kotor-vr-math-test PRIVATE deps/OpenXR-SDK/include)
target_compile_definitions(kotor-vr-math-test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(kotor-vr-math-test PRIVATE openxr_loader opengl32 user32 gdi32)
enable_testing()
add_test(NAME vr-matrices-and-cursor COMMAND kotor-vr-math-test)
